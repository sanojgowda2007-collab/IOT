void setup(){

Serial.begin(9600);

pinMode(Rled,OUTPUT);
pinMode(Gled,OUTPUT);
pinMode(Yled,OUTPUT);
pinMode(Bled,OUTPUT);

dht.begin();

WiFi.begin(ssid,password);

while(WiFi.status()!=WL_CONNECTED){
delay(500);
Serial.println("Connecting WiFi...");
}

Serial.println("WiFi Connected");
Serial.print("IP Address: ");
Serial.println(WiFi.localIP());

server.begin();
webSocket.begin();

}

void loop(){

webSocket.loop();

// SENSOR UPDATE
if(millis()-lastUpdate>=2000){

lastUpdate=millis();

float t=dht.readTemperature();
float h=dht.readHumidity();

if(!isnan(t)&&!isnan(h)){
String data=String(t,1)+","+String(h,1);
webSocket.broadcastTXT(data);
}
}

// WEB SERVER
WiFiClient client=server.available();

if(client){

String request=client.readStringUntil('\r');

// LED CONTROL
if(request.indexOf("/red/on")>=0)
digitalWrite(Rled,HIGH);

else if(request.indexOf("/red/off")>=0)
digitalWrite(Rled,LOW);

else if(request.indexOf("/green/on")>=0)
digitalWrite(Gled,HIGH);

else if(request.indexOf("/green/off")>=0)
digitalWrite(Gled,LOW);

else if(request.indexOf("/yellow/on")>=0)
digitalWrite(Yled,HIGH);

else if(request.indexOf("/yellow/off")>=0)
digitalWrite(Yled,LOW);

else if(request.indexOf("/blue/on")>=0)
digitalWrite(Bled,HIGH);

else if(request.indexOf("/blue/off")>=0)
digitalWrite(Bled,LOW);

// HTTP RESPONSE
client.println("HTTP/1.1 200 OK");
client.println("Content-Type: text/html");
client.println("Connection: close");
client.println();

client.print((const __FlashStringHelper*)webpage);

client.stop();

}

}
